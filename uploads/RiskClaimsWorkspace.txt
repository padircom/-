import { useEffect, useState } from "react";
import { type Bi, type Lang, t } from "../data/framework";

type Props = { lang: Lang; subId: string };
type WorkspaceMode = "risk" | "change" | "delay" | "claim";
type RiskStatus = "open" | "mitigated" | "accepted";

type RiskRow = {
  id: string;
  title: Bi;
  probability: number;
  impact: number;
  owner: string;
  status: RiskStatus;
};

type ChangeRow = {
  id: string;
  title: Bi;
  reason: Bi;
  timeImpact: string;
  costImpact: string;
  status: "pending" | "approved" | "rejected";
};

type DailyInput = {
  id: string;
  date: string;
  resourceLabel: string;
  note: string;
  updatedAt: string;
};

const modeFromSub = (subId: string): WorkspaceMode => {
  if (subId === "d4-p1-s2" || subId === "d4-p2-s1") return "risk"; // d4-p2-s1 kept for backward compatibility
  if (subId === "d4-p3-s1") return "change";
  if (subId === "d4-p4-s1") return "delay";
  if (subId === "d4-p5-s1") return "claim";
  return "risk";
};

const seedRisks: RiskRow[] = [
  { id: "r1", title: { fa: "تأخیر تأمین اقلام Long Lead", en: "Long-lead procurement delay" }, probability: 4, impact: 5, owner: "Procurement", status: "open" },
  { id: "r2", title: { fa: "محدودیت ماشین‌آلات سنگین کارگاه", en: "Heavy equipment constraint" }, probability: 4, impact: 4, owner: "Site Manager", status: "open" },
  { id: "r3", title: { fa: "تغییر مشخصات فنی توسط ذی‌نفع", en: "Stakeholder technical change" }, probability: 3, impact: 4, owner: "Engineering", status: "mitigated" },
  { id: "r4", title: { fa: "شرایط نامساعد جوی", en: "Adverse weather" }, probability: 2, impact: 3, owner: "Construction", status: "accepted" },
];

const seedChanges: ChangeRow[] = [
  { id: "cr1", title: { fa: "تغییر مسیر خط لوله اصلی", en: "Main pipeline route change" }, reason: { fa: "تعارض با زیرساخت موجود", en: "Conflict with existing infrastructure" }, timeImpact: "+12d", costImpact: "+$180k", status: "pending" },
  { id: "cr2", title: { fa: "افزایش کلاس بتن فونداسیون", en: "Foundation concrete class increase" }, reason: { fa: "الزام فنی جدید", en: "New technical requirement" }, timeImpact: "+4d", costImpact: "+$42k", status: "approved" },
];

const statusText: Record<RiskStatus, Bi> = {
  open: { fa: "باز", en: "Open" },
  mitigated: { fa: "کنترل‌شده", en: "Mitigated" },
  accepted: { fa: "پذیرفته‌شده", en: "Accepted" },
};

const storedDailyReports = (): DailyInput[] => {
  try {
    return JSON.parse(window.localStorage.getItem("pms:daily-reports") ?? "[]") as DailyInput[];
  } catch {
    return [];
  }
};

export default function RiskClaimsWorkspace({ lang, subId }: Props) {
  const rtl = lang === "fa";
  const mode = modeFromSub(subId);
  const [risks, setRisks] = useState<RiskRow[]>(seedRisks);
  const [changes, setChanges] = useState<ChangeRow[]>(seedChanges);
  const [dailyReports, setDailyReports] = useState<DailyInput[]>([]);
  const [aiOpen, setAiOpen] = useState(false);
  const [aiResult, setAiResult] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [riskTitle, setRiskTitle] = useState("");
  const [riskProbability, setRiskProbability] = useState("3");
  const [riskImpact, setRiskImpact] = useState("3");
  const [riskOwner, setRiskOwner] = useState("");
  const [changeTitle, setChangeTitle] = useState("");
  const [changeReason, setChangeReason] = useState("");
  const [privateTerms, setPrivateTerms] = useState("");
  const [evidence, setEvidence] = useState<string[]>([
    "Daily_Report / Daily_Report",
    "Correspondence_Master / Official Correspondence",
    "Engineering Document / Engineering Record",
  ]);
  const [evidenceInput, setEvidenceInput] = useState("");

  const labels = {
    risk: rtl ? "ثبت ریسک" : "Risk Register",
    analysis: rtl ? "تحلیل ریسک" : "Risk Analysis",
    change: rtl ? "مدیریت تغییرات" : "Change Management",
    delay: rtl ? "تحلیل تأخیرات" : "Delay Analysis",
    claim: rtl ? "مدیریت ادعاها" : "Claims Management",
  };

  useEffect(() => {
    setDailyReports(storedDailyReports());
  }, []);

  const refreshDailyReports = () => setDailyReports(storedDailyReports());
  const riskScore = (r: RiskRow) => r.probability * r.impact;
  const criticalRisks = risks.filter((r) => riskScore(r) >= 12).length;
  const totalDelayDays = 24 + dailyReports.length * 2;
  const recentDaily = dailyReports.slice(0, 4);

  const runAI = () => {
    setBusy(true);
    window.setTimeout(() => {
      let answer: string;
      if (mode === "risk") {
        answer = rtl
          ? `${criticalRisks} ریسک با امتیاز بحرانی شناسایی شد. اولویت اقدام روی تأمین اقلام Long Lead و محدودیت ماشین‌آلات است؛ مالک هر ریسک باید برنامه پاسخ و تاریخ بازبینی ثبت کند.`
          : `${criticalRisks} critical risks detected. Prioritize long-lead procurement and equipment constraints; each owner should register a response plan and review date.`;
      } else if (mode === "change") {
        answer = rtl
          ? "درخواست‌های تغییر بر اساس اثر زمان، هزینه و مدارک پشتیبان اولویت‌بندی شدند. قبل از تأیید، اثر تجمعی تغییرات بر Baseline و شرایط خصوصی پیمان بررسی شود."
          : "Change requests were prioritized by time, cost and supporting evidence. Before approval, assess cumulative impact on the baseline and Particular Conditions of Contract.";
      } else {
        answer = rtl
          ? `بر اساس ${dailyReports.length} گزارش روزانه، ${totalDelayDays} روز اثر تأخیر اولیه ثبت شده است. بخشنامه ۵۰۹۰، مواد ۲۹ و ۳۰، شرایط خصوصی پیمان، مکاتبات و مدارک مهندسی باید برای تعیین تأخیر مجاز/غیرمجاز تطبیق داده شوند.`
          : `Based on ${dailyReports.length} daily reports, ${totalDelayDays} preliminary delay days are recorded. Bulletin 5090, Articles 29 and 30, Particular Conditions, correspondence and engineering records must be reconciled to classify excusable/non-excusable delay.`;
      }
      setAiResult(answer);
      setAiOpen(true);
      setBusy(false);
    }, 850);
  };

  const addRisk = () => {
    if (!riskTitle.trim()) return;
    setRisks((prev) => [{ id: `r-${Date.now()}`, title: { fa: riskTitle, en: riskTitle }, probability: Number(riskProbability), impact: Number(riskImpact), owner: riskOwner || "PMO", status: "open" }, ...prev]);
    setRiskTitle("");
    setRiskOwner("");
  };

  const addChange = () => {
    if (!changeTitle.trim()) return;
    setChanges((prev) => [{ id: `cr-${Date.now()}`, title: { fa: changeTitle, en: changeTitle }, reason: { fa: changeReason || "ثبت نشده", en: changeReason || "Not specified" }, timeImpact: "TBD", costImpact: "TBD", status: "pending" }, ...prev]);
    setChangeTitle("");
    setChangeReason("");
  };

  const addEvidence = () => {
    if (!evidenceInput.trim()) return;
    setEvidence((prev) => [...prev, evidenceInput.trim()]);
    setEvidenceInput("");
  };

  return (
    <div className="mt-4 flex min-h-0 flex-1 flex-col gap-3 overflow-hidden" dir={rtl ? "rtl" : "ltr"}>
      <section className="glass-dark shrink-0 rounded-2xl px-3 py-2.5">
        <div className="flex flex-wrap items-center gap-2">
          <span className="grid h-8 w-8 place-items-center rounded-lg border border-rose-400/40 bg-rose-400/10 text-[14px]">{mode === "claim" || mode === "delay" ? "⚖" : "⚠"}</span>
          <div>
            <h3 className="text-[12px] font-normal tx1">{labels[mode]}</h3>
            <p className="text-[8.5px] font-extralight tx3">{rtl ? "محیط تخصصی بدون قالب کارفرما / مشاور" : "Specialist workspace without client / consultant templates"}</p>
          </div>
          <span className="ms-auto rounded-lg border border-fuchsia-400/45 bg-fuchsia-400/10 px-2.5 py-1 text-[9.5px] font-light text-fuchsia-300">✨ AI {rtl ? "فعال" : "Enabled"}</span>
          <button onClick={runAI} disabled={busy} className="rounded-lg border border-fuchsia-400/55 bg-fuchsia-400/12 px-2.5 py-1 text-[9.5px] font-light text-fuchsia-200 transition hover:bg-fuchsia-400/22 disabled:opacity-45">
            {busy ? (rtl ? "در حال تحلیل…" : "Analyzing…") : (rtl ? "مشاوره AI" : "AI Advisory")}
          </button>
        </div>
      </section>

      {mode === "risk" ? (
        <RiskWorkspace
          lang={lang}
          mode={mode}
          risks={risks}
          riskTitle={riskTitle}
          setRiskTitle={setRiskTitle}
          riskProbability={riskProbability}
          setRiskProbability={setRiskProbability}
          riskImpact={riskImpact}
          setRiskImpact={setRiskImpact}
          riskOwner={riskOwner}
          setRiskOwner={setRiskOwner}
          addRisk={addRisk}
          runAI={runAI}
          criticalRisks={criticalRisks}
        />
      ) : mode === "change" ? (
        <ChangeWorkspace
          lang={lang}
          changes={changes}
          title={changeTitle}
          setTitle={setChangeTitle}
          reason={changeReason}
          setReason={setChangeReason}
          addChange={addChange}
          runAI={runAI}
        />
      ) : (
        <DelayClaimWorkspace
          lang={lang}
          mode={mode}
          dailyReports={dailyReports}
          recentDaily={recentDaily}
          refreshDailyReports={refreshDailyReports}
          privateTerms={privateTerms}
          setPrivateTerms={setPrivateTerms}
          evidence={evidence}
          evidenceInput={evidenceInput}
          setEvidenceInput={setEvidenceInput}
          addEvidence={addEvidence}
          runAI={runAI}
          totalDelayDays={totalDelayDays}
        />
      )}

      {aiOpen && (
        <div className="fade-rise shrink-0 rounded-2xl border border-fuchsia-400/40 bg-fuchsia-400/8 p-3">
          <div className="flex items-center gap-2">
            <span>✨</span>
            <span className="text-[10.5px] font-normal text-fuchsia-200">{rtl ? "مشاوره هوش مصنوعی" : "AI Advisory"}</span>
            <button onClick={() => setAiOpen(false)} className="ms-auto text-fuchsia-200/70 hover:text-fuchsia-100">✕</button>
          </div>
          <p className="mt-2 text-[10px] font-light leading-5 text-fuchsia-100">{aiResult}</p>
        </div>
      )}

      <footer className="b-line-soft flex shrink-0 items-center gap-2 border-t pt-2">
        <span className="pulse-dot h-[7px] w-[7px] rounded-full bg-emerald-400" />
        <span className="text-[9px] font-extralight tx3">{rtl ? "ذخیره‌سازی و سوابق:" : "Storage & records:"}</span>
        <span className="text-[9px] font-light ok-dim-t" dir="ltr">SQL Server (.\SQL2008EXPRESS)</span>
        <span className="ms-auto text-[8.5px] font-extralight tx4">{rtl ? "منبع روزانه + قرارداد + مکاتبات + مدارک مهندسی" : "Daily reports + contract + correspondence + engineering records"}</span>
      </footer>
    </div>
  );
}

function RiskWorkspace({
  lang, mode, risks, riskTitle, setRiskTitle, riskProbability, setRiskProbability, riskImpact, setRiskImpact, riskOwner, setRiskOwner, addRisk, runAI, criticalRisks,
}: {
  lang: Lang;
  mode: "risk" | "analysis";
  risks: RiskRow[];
  riskTitle: string;
  setRiskTitle: (v: string) => void;
  riskProbability: string;
  setRiskProbability: (v: string) => void;
  riskImpact: string;
  setRiskImpact: (v: string) => void;
  riskOwner: string;
  setRiskOwner: (v: string) => void;
  addRisk: () => void;
  runAI: () => void;
  criticalRisks: number;
}) {
  const rtl = lang === "fa";
  const [p, setP] = useState(3);
  const [i, setI] = useState(3);
  const [hoverCell, setHoverCell] = useState<string | null>(null);
  const cellColor = (probability: number, impact: number) => {
    const score = probability * impact;
    return score >= 16 ? "#EF4444" : score >= 9 ? "#F59E0B" : "#10B981";
  };
  return (
    <div className="grid min-h-0 flex-1 grid-cols-1 gap-3 overflow-hidden xl:grid-cols-[minmax(330px,1.05fr)_minmax(0,1.25fr)]">
      <section className="glass-dark flex min-h-0 flex-col overflow-hidden rounded-2xl">
        <header className="border-b b-line-soft px-3 py-2.5">
          <h4 className="text-[11px] font-normal tx1">{rtl ? "ثبت داخلی ریسک" : "Internal Risk Register"}</h4>
          <p className="text-[8.5px] font-extralight tx3">{mode === "analysis" ? (rtl ? "تحلیل احتمال × اثر" : "Probability × impact analysis") : (rtl ? "بدون قالب ابلاغی" : "No mandated template")}</p>
        </header>
        <div className="grid grid-cols-2 gap-2 border-b b-line-soft p-2.5">
          <input value={riskTitle} onChange={(e) => setRiskTitle(e.target.value)} placeholder={rtl ? "عنوان ریسک…" : "Risk title…"} className="col-span-2 rounded-lg border b-line-soft bg-[var(--row)] px-2 py-1.5 text-[10px] tx1 outline-none" />
          <select value={riskProbability} onChange={(e) => setRiskProbability(e.target.value)} className="rounded-lg border b-line-soft bg-[var(--row)] px-2 py-1.5 text-[9.5px] tx1 outline-none"><option value="1">{rtl ? "احتمال ۱" : "Probability 1"}</option><option value="2">{rtl ? "احتمال ۲" : "Probability 2"}</option><option value="3">{rtl ? "احتمال ۳" : "Probability 3"}</option><option value="4">{rtl ? "احتمال ۴" : "Probability 4"}</option><option value="5">{rtl ? "احتمال ۵" : "Probability 5"}</option></select>
          <select value={riskImpact} onChange={(e) => setRiskImpact(e.target.value)} className="rounded-lg border b-line-soft bg-[var(--row)] px-2 py-1.5 text-[9.5px] tx1 outline-none"><option value="1">{rtl ? "اثر ۱" : "Impact 1"}</option><option value="2">{rtl ? "اثر ۲" : "Impact 2"}</option><option value="3">{rtl ? "اثر ۳" : "Impact 3"}</option><option value="4">{rtl ? "اثر ۴" : "Impact 4"}</option><option value="5">{rtl ? "اثر ۵" : "Impact 5"}</option></select>
          <input value={riskOwner} onChange={(e) => setRiskOwner(e.target.value)} placeholder={rtl ? "مالک ریسک…" : "Risk owner…"} className="rounded-lg border b-line-soft bg-[var(--row)] px-2 py-1.5 text-[10px] tx1 outline-none" />
          <button onClick={addRisk} className="rounded-lg border border-rose-400/50 bg-rose-400/10 px-2 py-1.5 text-[9.5px] font-light text-rose-300">+ {rtl ? "ثبت ریسک" : "Add risk"}</button>
        </div>
        <div className="thin-scroll min-h-0 flex-1 space-y-1.5 overflow-y-auto p-2">
          {risks.map((risk) => <div key={risk.id} className="glass-row rounded-xl px-2.5 py-2"><div className="flex items-center gap-2"><i className="h-2 w-2 rounded-full" style={{ background: cellColor(risk.probability, risk.impact) }} /><span className="min-w-0 flex-1 truncate text-[10px] font-light tx1">{t(risk.title, lang)}</span><b className="text-[10px] tabular-nums" style={{ color: cellColor(risk.probability, risk.impact) }}>{risk.probability * risk.impact}</b></div><div className="mt-1 flex items-center gap-2 text-[8.5px] font-extralight tx3"><span>{statusText[risk.status][lang]}</span><span>·</span><span>{risk.owner}</span><span className="ms-auto">P{risk.probability} × I{risk.impact}</span></div></div>)}
        </div>
      </section>
      <section className="glass-dark flex min-h-0 flex-col overflow-hidden rounded-2xl p-3">
        <div className="flex items-center gap-2"><h4 className="text-[11px] font-normal tx1">{rtl ? "ماتریس حرارتی احتمال / اثر" : "Probability / Impact Heatmap"}</h4><span className="ms-auto text-[9px] font-light text-rose-300">{criticalRisks} {rtl ? "بحرانی" : "critical"}</span></div>
        <div className="mt-3 grid grid-cols-[22px_repeat(5,minmax(0,1fr))] gap-1">
          <div />{[1, 2, 3, 4, 5].map((n) => <div key={n} className="text-center text-[8px] tx4">I{n}</div>)}
          {[5, 4, 3, 2, 1].map((probability) => <div key={probability} className="contents"><div className="flex items-center justify-center text-[8px] tx4">P{probability}</div>{[1, 2, 3, 4, 5].map((impact) => { const key = `${probability}-${impact}`; const count = risks.filter((r) => r.probability === probability && r.impact === impact).length; return <button key={key} onMouseEnter={() => { setP(probability); setI(impact); setHoverCell(key); }} onMouseLeave={() => setHoverCell(null)} className="relative aspect-square rounded-lg border transition hover:scale-105" style={{ background: `${cellColor(probability, impact)}30`, borderColor: `${cellColor(probability, impact)}88` }}><span className="text-[10px] font-medium" style={{ color: cellColor(probability, impact) }}>{count || "·"}</span>{hoverCell === key && <span className="absolute inset-x-0 -top-5 z-10 text-[7px] tx1">{p} × {i}</span>}</button>; })}</div>)}
        </div>
        <div className="mt-3 flex items-center justify-between text-[8.5px] tx3"><span><i className="me-1 inline-block h-2 w-2 rounded-full bg-emerald-400" />{rtl ? "کم" : "Low"}</span><span><i className="me-1 inline-block h-2 w-2 rounded-full bg-amber-400" />{rtl ? "متوسط" : "Medium"}</span><span><i className="me-1 inline-block h-2 w-2 rounded-full bg-rose-500" />{rtl ? "بحرانی" : "Critical"}</span></div>
        <div className="mt-4 rounded-xl border border-fuchsia-400/30 bg-fuchsia-400/8 p-3"><div className="text-[10px] font-normal text-fuchsia-200">✨ {rtl ? "تحلیل AI" : "AI Analysis"}</div><p className="mt-1 text-[9.5px] leading-5 tx2">{rtl ? "ماتریس بر اساس احتمال و اثر به‌روزرسانی می‌شود. برای پیشنهاد پاسخ به ریسک، مشاوره AI را اجرا کنید." : "The matrix updates from probability and impact. Run AI Advisory for response recommendations."}</p><button onClick={runAI} className="mt-2 rounded-lg border border-fuchsia-400/50 bg-fuchsia-400/12 px-2.5 py-1 text-[9px] text-fuchsia-200">{rtl ? "اجرای تحلیل" : "Run analysis"}</button></div>
      </section>
    </div>
  );
}

  const rtl = lang === "fa";
  return <div className="grid min-h-0 flex-1 grid-cols-1 gap-3 overflow-hidden xl:grid-cols-[minmax(320px,.9fr)_minmax(0,1.4fr)]"><section className="glass-dark flex min-h-0 flex-col rounded-2xl p-3"><h4 className="text-[11px] font-normal tx1">{rtl ? "ثبت درخواست تغییر داخلی" : "Internal Change Request"}</h4><p className="mt-0.5 text-[8.5px] tx3">{rtl ? "تحلیل اثر بدون قالب کارفرما" : "Impact analysis without a client template"}</p><div className="mt-3 space-y-2"><input value={title} onChange={(e) => setTitle(e.target.value)} placeholder={rtl ? }

function DelayClaimWorkspace({ lang, mode, dailyReports, recentDaily, refreshDailyReports, privateTerms, setPrivateTerms, evidence, evidenceInput, setEvidenceInput, addEvidence, runAI, totalDelayDays }: { lang: Lang; mode: "delay" | "claim"; dailyReports: DailyInput[]; recentDaily: DailyInput[]; refreshDailyReports: () => void; privateTerms: string; setPrivateTerms: (v: string) => void; evidence: string[]; evidenceInput: string; setEvidenceInput: (v: string) => void; addEvidence: () => void; runAI: () => void; totalDelayDays: number }) {
  const rtl = lang === "fa";
  return <div className="grid min-h-0 flex-1 grid-cols-1 gap-3 overflow-hidden xl:grid-cols-[minmax(330px,.9fr)_minmax(0,1.35fr)]"><section className="glass-dark flex min-h-0 flex-col overflow-hidden rounded-2xl"><header className="border-b b-line-soft px-3 py-2.5"><h4 className="text-[11px] font-normal tx1">{mode === "delay" ?
