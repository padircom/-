import { useState } from "react";
import { t, type Bi, type Lang } from "../data/framework";

type EdmsTab = "mdr" | "correspondence" | "transmittal" | "lessons";

type DocumentRow = {
  id: string;
  code: string;
  title: Bi;
  revision: string;
  discipline: string;
  status: "approved" | "under_review" | "rejected" | "draft";
  updatedAt: string;
};

type LetterRow = {
  id: string;
  letterNo: string;
  subject: Bi;
  sender: Bi;
  receiver: Bi;
  date: string;
  actionRequired: boolean;
  dueDate: string;
};

type TransmittalRow = {
  id: string;
  transmittalNo: string;
  purpose: Bi;
  recipient: Bi;
  docCount: number;
  date: string;
  status: "sent" | "received" | "ack";
};

type LessonRow = {
  id: string;
  title: Bi;
  category: Bi;
  impact: Bi;
  lesson: Bi;
  author: string;
};

const sampleDocs: DocumentRow[] = [
  { id: "doc1", code: "OG-2401-CIV-DR-001", title: { fa: "نقشه تفصیلی فونداسیون مخازن", en: "Foundation Detailed Drawing" }, revision: "Rev-02", discipline: "Civil", status: "approved", updatedAt: "1403/02/10" },
  { id: "doc2", code: "OG-2401-MEC-DS-004", title: { fa: "برگ مشخصات فنی پمپ‌های سانتریفیوژ", en: "Centrifugal Pump Data Sheet" }, revision: "Rev-01", discipline: "Mechanical", status: "under_review", updatedAt: "1403/02/15" },
  { id: "doc3", code: "OG-2401-PIP-ISO-012", title: { fa: "نقشه ایزومتریک خطوط لوله کشی فاز ۱", en: "Piping Isometric Line 012" }, revision: "Rev-00", discipline: "Piping", status: "draft", updatedAt: "1403/02/18" },
];

const sampleLetters: LetterRow[] = [
  { id: "let1", letterNo: "LTR-2401-104", subject: { fa: "درخواست ابلاغ نقشه‌های اصلاحی کیلومتر ۲۴", en: "Request for revised drawings KM24" }, sender: { fa: "پیمانکار", en: "Contractor" }, receiver: { fa: "مشاور", en: "Consultant" }, date: "1403/02/12", actionRequired: true, dueDate: "1403/02/20" },
  { id: "let2", letterNo: "LTR-2401-089", subject: { fa: "پاسخ به ادعای شرایط نامساعد جوی", en: "Response to weather claim" }, sender: { fa: "کارفرما", en: "Employer" }, receiver: { fa: "پیمانکار", en: "Contractor" }, date: "1403/02/05", actionRequired: false, dueDate: "-" },
];

const sampleTransmittals: TransmittalRow[] = [
  { id: "tr1", transmittalNo: "TR-OG-2401-042", purpose: { fa: "جهت بررسی و تأیید", en: "For Review & Approval" }, recipient: { fa: "دستگاه نظارت", en: "Supervision Team" }, docCount: 6, date: "1403/02/14", status: "sent" },
  { id: "tr2", transmittalNo: "TR-OG-2401-038", purpose: { fa: "جهت ساخت و اجرا", en: "For Construction (IFC)" }, recipient: { fa: "پیمانکار اجرایی", en: "Contractor" }, docCount: 12, date: "1403/02/01", status: "ack" },
];

const sampleLessons: LessonRow[] = [
  { id: "ls1", title: { fa: "تأخیر در تأیید نقشه‌های شاپ قالب‌بندی", en: "Delay in shop drawing approvals" }, category: { fa: "مهندسی", en: "Engineering" }, impact: { fa: "تأخیر ۲ هفته‌ای در بتن‌ریزی", en: "2-week delay in concreting" }, lesson: { fa: "ارسال همزمان نسخه‌های الکترونیکی جهت تسریع بررسی پیش از جلسه حضوری", en: "Parallel electronic submission prior to formal review" }, author: "مهندس احمدی" },
  { id: "ls2", title: { fa: "تعارض خطوط لوله زیرزمینی با کابل فشار قوی", en: "Underground pipe & HV cable conflict" }, category: { fa: "اجرا / سایت", en: "Construction" }, impact: { fa: "توقف ۲ روزه حفاری", en: "2-day excavation stoppage" }, lesson: { fa: "انجام اسکن سونار کارگاهی قبل از گودبرداری در زون‌های صنعتی قدیمی", en: "Perform site sonar scan prior to excavation in old zones" }, author: "مهندس رضایی" },
];

const statusMetaDoc = {
  approved: { fa: "تأییدشده", en: "Approved", color: "#34D399" },
  under_review: { fa: "در حال بررسی", en: "Under Review", color: "#FBBF24" },
  rejected: { fa: "ردشده", en: "Rejected", color: "#F87171" },
  draft: { fa: "پیش‌نویس", en: "Draft", color: "#94A3B8" },
};

export default function DocumentWorkspace({ lang }: { lang: Lang }) {
  const rtl = lang === "fa";
  const [tab, setTab] = useState<EdmsTab>("mdr");
  const [docs, setDocs] = useState(sampleDocs);
  const [letters] = useState(sampleLetters);
  const [transmittals] = useState(sampleTransmittals);
  const [lessons] = useState(sampleLessons);

  // Modal / Add form state
  const [newDocCode, setNewDocCode] = useState("");
  const [newDocTitle, setNewDocCodeTitle] = useState("");
  const [newDocDiscipline, setNewDocDiscipline] = useState("Civil");

  const addDocument = () => {
    if (!newDocCode.trim()) return;
    setDocs((prev) => [
      {
        id: `doc-${Date.now()}`,
        code: newDocCode,
        title: { fa: newDocTitle || "مدرک جدید", en: newDocTitle || "New Document" },
        revision: "Rev-00",
        discipline: newDocDiscipline,
        status: "under_review",
        updatedAt: new Date().toLocaleDateString(rtl ? "fa-IR" : "en-GB"),
      },
      ...prev,
    ]);
    setNewDocCode("");
    setNewDocCodeTitle("");
  };

  return (
    <div className="mt-4 flex min-h-0 flex-1 flex-col gap-3 overflow-hidden" dir={rtl ? "rtl" : "ltr"}>
      {/* Top Header */}
      <section className="glass-dark shrink-0 rounded-2xl p-3">
        <div className="flex flex-wrap items-center gap-2">
          <span className="grid h-8 w-8 place-items-center rounded-lg border border-sky-400/40 bg-sky-400/10 text-[15px]">🗂</span>
          <div>
            <h3 className="text-[12px] font-semibold tx1">{rtl ? "سیستم الکترونیکی مدیریت اسناد و مدارک (EDMS)" : "Electronic Document Management System"}</h3>
            <p className="text-[8.5px] font-extralight tx3">{rtl ? "کنترل مدارک مهندسی (MDR)، مکاتبات، ترانسمیتال‌ها و بانک درس‌آموخته‌ها" : "MDR, correspondence, transmittals & lessons learned repository"}</p>
          </div>
          <span className="ms-auto font-mono text-[9px] text-sky-300">dbo.Document_Master · dbo.Transmittal_Register</span>
        </div>
      </section>

      {/* Tabs */}
      <nav className="flex shrink-0 flex-wrap items-center gap-1.5 rounded-xl bg-black/15 p-1">
        {[
          { id: "mdr" as const, fa: "مدارک مهندسی (MDR)", en: "Engineering MDR", icon: "📐" },
          { id: "correspondence" as const, fa: "مکاتبات و نامه‌ها", en: "Correspondence", icon: "✉️" },
          { id: "transmittal" as const, fa: "ترانسمیتال‌ها", en: "Transmittals", icon: "📦" },
          { id: "lessons" as const, fa: "درس‌آموخته‌ها", en: "Lessons Learned", icon: "💡" },
        ].map((tItem) => (
          <button
            key={tItem.id}
            onClick={() => setTab(tItem.id)}
            className={`flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[10px] font-light transition ${
              tab === tItem.id ? "toggle-on tx1 shadow-sm" : "tx3 hover:tx2"
            }`}
          >
            <span>{tItem.icon}</span>
            <span>{rtl ? tItem.fa : tItem.en}</span>
          </button>
        ))}
      </nav>

      {/* Main Tab View */}
      <div className="thin-scroll min-h-0 flex-1 overflow-y-auto pr-1">
        {/* TAB 1: MDR */}
        {tab === "mdr" && (
          <div className="fade-rise space-y-3">
            {/* Quick Add */}
            <div className="glass-dark flex flex-wrap items-center gap-2 rounded-xl p-2.5">
              <span className="text-[10px] font-normal tx1">{rtl ? "ثبت مدرک جدید در MDR:" : "Register New Document:"}</span>
              <input value={newDocCode} onChange={(e) => setNewDocCode(e.target.value)} placeholder="کد مدرک (مثال: OG-2401-ELE-DS-002)" dir="ltr" className="w-48 rounded-lg border b-line-soft bg-black/20 px-2 py-1 text-[10px] tx1 outline-none" />
              <input value={newDocTitle} onChange={(e) => setNewDocCodeTitle(e.target.value)} placeholder="عنوان مدرک…" className="w-56 rounded-lg border b-line-soft bg-black/20 px-2 py-1 text-[10px] tx1 outline-none" />
              <select value={newDocDiscipline} onChange={(e) => setNewDocDiscipline(e.target.value)} className="rounded-lg border b-line-soft bg-black/20 px-2 py-1 text-[10px] tx1 outline-none" style={{ colorScheme: "dark" }}>
                <option value="Civil">Civil</option>
                <option value="Mechanical">Mechanical</option>
                <option value="Piping">Piping</option>
                <option value="Electrical">Electrical</option>
                <option value="Instrumentation">Instrumentation</option>
                <option value="Process">Process</option>
              </select>
              <button onClick={addDocument} className="rounded-lg border border-sky-400/50 bg-sky-400/15 px-3 py-1 text-[10px] font-light text-sky-200 hover:bg-sky-400/25">+ {rtl ? "ثبت مدرک" : "Add"}</button>
            </div>

            {/* MDR Table */}
            <div className="glass-dark overflow-x-auto rounded-2xl p-3">
              <table className="w-full min-w-[720px] border-collapse text-[10px]">
                <thead>
                  <tr className="border-b b-line-soft bg-black/25 text-[9px] font-extralight tx3">
                    <th className="px-2 py-2 text-start">{rtl ? "کد مدرک" : "Doc Code"}</th>
                    <th className="px-2 py-2 text-start">{rtl ? "عنوان مدرک" : "Document Title"}</th>
                    <th className="px-2 py-2 text-center">{rtl ? "دیسپلین" : "Discipline"}</th>
                    <th className="px-2 py-2 text-center">{rtl ? "نسخه" : "Revision"}</th>
                    <th className="px-2 py-2 text-center">{rtl ? "وضعیت بررسی" : "Status"}</th>
                    <th className="px-2 py-2 text-center">{rtl ? "تاریخ بروزرسانی" : "Updated"}</th>
                  </tr>
                </thead>
                <tbody className="divide-y b-line-soft">
                  {docs.map((doc) => {
                    const st = statusMetaDoc[doc.status];
                    return (
                      <tr key={doc.id} className="hover:bg-white/[0.02]">
                        <td className="px-2 py-1.5 font-mono text-[9.5px] tx2" dir="ltr">{doc.code}</td>
                        <td className="px-2 py-1.5 font-normal tx1">{t(doc.title, lang)}</td>
                        <td className="px-2 py-1.5 text-center tx3">{doc.discipline}</td>
                        <td className="px-2 py-1.5 text-center font-mono text-[9px] text-sky-300" dir="ltr">{doc.revision}</td>
                        <td className="px-2 py-1.5 text-center">
                          <span className="inline-flex items-center gap-1 rounded px-2 py-0.5 text-[8.5px]" style={{ background: `${st.color}18`, color: st.color }}>
                            <i className="h-1.5 w-1.5 rounded-full" style={{ background: st.color }} />
                            {t({ fa: st.fa, en: st.en }, lang)}
                          </span>
                        </td>
                        <td className="px-2 py-1.5 text-center font-mono text-[9px] tx3" dir="ltr">{doc.updatedAt}</td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* TAB 2: CORRESPONDENCE */}
        {tab === "correspondence" && (
          <div className="fade-rise glass-dark overflow-x-auto rounded-2xl p-3 space-y-2">
            <div className="flex items-center justify-between border-b b-line-soft pb-2">
              <span className="text-[11px] font-normal tx1">{rtl ? "دفتر ثبت مکاتبات و نامه‌ها" : "Correspondence Register"}</span>
              <span className="text-[8.5px] font-extralight tx3">Action Register</span>
            </div>
            <table className="w-full min-w-[760px] border-collapse text-[10px]">
              <thead>
                <tr className="border-b b-line-soft bg-black/25 text-[9px] font-extralight tx3">
                  <th className="px-2 py-2 text-start">{rtl ? "شماره نامه" : "Letter No"}</th>
                  <th className="px-2 py-2 text-start">{rtl ? "موضوع نامه" : "Subject"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "فرستنده" : "Sender"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "گیرنده" : "Receiver"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "تاریخ" : "Date"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "اقدام لازم" : "Action Required"}</th>
                </tr>
              </thead>
              <tbody className="divide-y b-line-soft">
                {letters.map((letRow) => (
                  <tr key={letRow.id} className="hover:bg-white/[0.02]">
                    <td className="px-2 py-1.5 font-mono text-[9.5px] tx2" dir="ltr">{letRow.letterNo}</td>
                    <td className="px-2 py-1.5 font-normal tx1">{t(letRow.subject, lang)}</td>
                    <td className="px-2 py-1.5 text-center tx3">{t(letRow.sender, lang)}</td>
                    <td className="px-2 py-1.5 text-center tx3">{t(letRow.receiver, lang)}</td>
                    <td className="px-2 py-1.5 text-center font-mono text-[9px] tx3" dir="ltr">{letRow.date}</td>
                    <td className="px-2 py-1.5 text-center">
                      {letRow.actionRequired ? (
                        <span className="rounded border border-amber-400/40 bg-amber-400/15 px-2 py-0.5 text-[8.5px] text-amber-300">
                          ⚡ {rtl ? "مهلت:" : "Due:"} {letRow.dueDate}
                        </span>
                      ) : (
                        <span className="text-[8.5px] tx4">{rtl ? "اطلاعی" : "Info only"}</span>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}

        {/* TAB 3: TRANSMITTALS */}
        {tab === "transmittal" && (
          <div className="fade-rise glass-dark overflow-x-auto rounded-2xl p-3 space-y-2">
            <div className="flex items-center justify-between border-b b-line-soft pb-2">
              <span className="text-[11px] font-normal tx1">{rtl ? "دفتر ترانسمیتال‌ها (ارسال و دریافت مدارک)" : "Transmittals Register"}</span>
            </div>
            <table className="w-full min-w-[680px] border-collapse text-[10px]">
              <thead>
                <tr className="border-b b-line-soft bg-black/25 text-[9px] font-extralight tx3">
                  <th className="px-2 py-2 text-start">{rtl ? "شماره ترانسمیتال" : "Transmittal No"}</th>
                  <th className="px-2 py-2 text-start">{rtl ? "هدف از ارسال" : "Purpose"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "تحویل‌گیرنده" : "Recipient"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "تعداد مدارک" : "Doc Count"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "تاریخ" : "Date"}</th>
                  <th className="px-2 py-2 text-center">{rtl ? "وضعیت رسید" : "Status"}</th>
                </tr>
              </thead>
              <tbody className="divide-y b-line-soft">
                {transmittals.map((tr) => (
                  <tr key={tr.id} className="hover:bg-white/[0.02]">
                    <td className="px-2 py-1.5 font-mono text-[9.5px] tx2" dir="ltr">{tr.transmittalNo}</td>
                    <td className="px-2 py-1.5 font-normal tx1">{t(tr.purpose, lang)}</td>
                    <td className="px-2 py-1.5 text-center tx3">{t(tr.recipient, lang)}</td>
                    <td className="px-2 py-1.5 text-center font-mono text-[9.5px] text-sky-300">{tr.docCount}</td>
                    <td className="px-2 py-1.5 text-center font-mono text-[9px] tx3" dir="ltr">{tr.date}</td>
                    <td className="px-2 py-1.5 text-center">
                      <span className={`rounded px-2 py-0.5 text-[8.5px] ${tr.status === "ack" ? "bg-emerald-400/15 text-emerald-300" : "bg-sky-400/15 text-sky-300"}`}>
                        {tr.status === "ack" ? (rtl ? "✓ رسید شده" : "Acknowledged") : (rtl ? "ارسال‌شده" : "Sent")}
                      </span>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}

        {/* TAB 4: LESSONS LEARNED */}
        {tab === "lessons" && (
          <div className="fade-rise space-y-2.5">
            {lessons.map((ls) => (
              <div key={ls.id} className="glass-dark rounded-2xl p-3.5 space-y-1.5 border border-sky-400/30">
                <div className="flex items-center justify-between">
                  <span className="text-[11.5px] font-medium text-sky-300">💡 {t(ls.title, lang)}</span>
                  <span className="rounded bg-sky-400/10 px-2 py-0.5 text-[8.5px] text-sky-200">{t(ls.category, lang)}</span>
                </div>
                <div className="text-[10px] tx3"><b className="tx2">{rtl ? "اثر و پیامد:" : "Impact:"}</b> {t(ls.impact, lang)}</div>
                <div className="rounded-xl border b-line-soft bg-black/15 p-2.5 text-[10.5px] font-light tx1 leading-5">
                  <b className="text-emerald-300">{rtl ? "درس‌آموخته جهت اقدام:" : "Lesson Learned:"}</b> {t(ls.lesson, lang)}
                </div>
                <div className="text-end text-[8.5px] tx4">{rtl ? "ثبت‌کننده:" : "Author:"} {ls.author}</div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
